          import pathlib
          import re
          import sys

          root = pathlib.Path(sys.argv[1])
          patterns = [
              ("GitHub token", re.compile(r"\b(?:gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{20,})\b")),
              ("AWS access key", re.compile(r"\bAKIA[0-9A-Z]{16}\b")),
              ("Google API key", re.compile(r"\bAIza[0-9A-Za-z_-]{30,}\b")),
              ("private key", re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----")),
              ("hard-coded credential", re.compile(
                  r"""(?i)\b(?:api[_-]?key|access[_-]?token|client[_-]?secret|password)\b\s*[:=]\s*["'](?!\s*(?:CHANGE_ME|YOUR_|<|\.\.\.))[\w./+=-]{24,}["']"""
              )),
          ]

          findings = []
          for path in root.rglob("*"):
              if not path.is_file():
                  continue
              try:
                  text = path.read_text(encoding="utf-8")
              except (UnicodeDecodeError, OSError):
                  continue
              for line_no, line in enumerate(text.splitlines(), 1):
                  for label, pattern in patterns:
                      if pattern.search(line):
                          findings.append((path.relative_to(root), line_no, label))

          if findings:
              for path, line_no, label in findings:

      - name: Validate token against target repository
        env:
          MIRROR_TOKEN: ${{ secrets.DEEPSEEK_MIRROR_TOKEN }}
        run: |
          set -euo pipefail
          RESPONSE_FILE="$RUNNER_TEMP/target-repo-response.json"
          HTTP_STATUS="$(curl --silent --show-error --output "$RESPONSE_FILE" --write-out '%{http_code}' \
            --header "Authorization: Bearer $MIRROR_TOKEN" \
            --header "Accept: application/vnd.github+json" \
            --header "X-GitHub-Api-Version: 2022-11-28" \
            https://api.github.com/repos/artdota32-boop/raid_app_deepseek)"

          case "$HTTP_STATUS" in
            200) ;;
            401)
              echo "::error::GitHub rejected the mirror token. It may be invalid, expired, or revoked."
              exit 1
              ;;
            403)
              echo "::error::GitHub accepted the token identity but denied access. Check repository permissions or token policy."
              exit 1
              ;;
            404)
              echo "::error::The token cannot access raid_app_deepseek, or the repository name is incorrect."
              exit 1
              ;;
            *)
              echo "::error::GitHub API returned HTTP $HTTP_STATUS while validating the mirror token."
              cat "$RESPONSE_FILE"

          import json
          import sys

          with open(sys.argv[1], encoding="utf-8") as f:

          if not permissions.get("push", False):
              print("::error::The token can read the repository but does not have push permission.")
              sys.exit(1)

          print("Mirror token is valid and has push access to raid_app_deepseek.")
